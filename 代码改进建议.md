```python
# network_analyzer.py
from igraph import Graph
from pathlib import Path
from typing import List, Dict, Optional

class NetworkAnalyzer:
    """网络分析器基类"""

    def __init__(self, network_file: str):
        """
        初始化网络分析器

        Args:
            network_file: 网络文件路径（Pajek 格式）
        """
        self.network_file = Path(network_file)
        self.graph: Optional[Graph] = None
        self.communities = None

    def load_network(self) -> Graph:
        """加载网络文件"""
        if not self.network_file.exists():
            raise FileNotFoundError(f"Network file not found: {self.network_file}")

        self.graph = Graph.Read_Pajek(str(self.network_file))
        self.graph = self.graph.simplify()
        return self.graph

    def detect_communities(self, method: str = 'walktrap') -> Dict:
        """
        检测社区结构

        Args:
            method: 算法选择 ('walktrap', 'eigenvector', 'betweenness')

        Returns:
            包含社区信息和模块度的字典
        """
        if self.graph is None:
            self.load_network()

        if method == 'walktrap':
            result = self.graph.community_walktrap()
        elif method == 'eigenvector':
            result = self.graph.community_leading_eigenvector()
        elif method == 'betweenness':
            result = self.graph.community_edge_betweenness()
        else:
            raise ValueError(f"Unknown method: {method}")

        self.communities = result.as_clustering()

        return {
            'membership': self.communities.membership,
            'modularity': self.communities.modularity,
            'num_communities': len(self.communities)
        }

    def compare_algorithms(self, methods: List[str] = None) -> Dict:
        """
        比较多种社区检测算法

        Args:
            methods: 要比较的算法列表

        Returns:
            各算法的结果对比
        """
        if methods is None:
            methods = ['walktrap', 'eigenvector', 'betweenness']

        results = {}
        for method in methods:
            try:
                results[method] = self.detect_communities(method)
            except Exception as e:
                print(f"Error with {method}: {e}")
                results[method] = None

        return results

    def visualize(self, output_file: str = None, layout: str = 'fr'):
        """
        可视化网络

        Args:
            output_file: 输出文件路径
            layout: 布局算法 ('fr', 'circle', 'kk')
        """
        if self.graph is None:
            self.load_network()

        # 如果已检测社区，使用颜色标记
        if self.communities is not None:
            colors = self._get_colors(len(self.communities))
            self.graph.vs['color'] = [colors[m] for m in self.communities.membership]

        # 绘制图形
        visual_style = {
            'layout': self.graph.layout(layout),
            'vertex_size': 10,
            'bbox': (800, 800),
            'margin': 50
        }

        if output_file:
            self.graph.write_svg(output_file, **visual_style)
        else:
            from igraph import plot
            plot(self.graph, **visual_style)

    @staticmethod
    def _get_colors(n: int) -> List[str]:
        """生成 n 种不同的颜色"""
        import colorsys
        colors = []
        for i in range(n):
            hue = i / n
            rgb = colorsys.hsv_to_rgb(hue, 0.8, 0.9)
            colors.append('#{:02x}{:02x}{:02x}'.format(
                int(rgb[0]*255), int(rgb[1]*255), int(rgb[2]*255)
            ))
        return colors

    def get_statistics(self) -> Dict:
        """获取网络统计信息"""
        if self.graph is None:
            self.load_network()

        return {
            'num_vertices': self.graph.vcount(),
            'num_edges': self.graph.ecount(),
            'density': self.graph.density(),
            'is_connected': self.graph.is_connected(),
            'avg_degree': sum(self.graph.degree()) / self.graph.vcount(),
            'diameter': self.graph.diameter() if self.graph.is_connected() else None
        }


# 使用示例
if __name__ == "__main__":
    # 创建分析器
    analyzer = NetworkAnalyzer("net/ws300.net")

    # 加载网络
    analyzer.load_network()

    # 获取统计信息
    stats = analyzer.get_statistics()
    print(f"网络统计: {stats}")

    # 比较算法
    results = analyzer.compare_algorithms()
    for method, result in results.items():
        if result:
            print(f"{method}: 模块度 = {result['modularity']:.4f}, "
                  f"社区数 = {result['num_communities']}")

    # 可视化
    analyzer.visualize(output_file="network_visualization.svg")
```

```python
import logging
from pathlib import Path

# 配置日志
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('network_analysis.log'),
        logging.StreamHandler()
    ]
)

logger = logging.getLogger(__name__)

def load_network(filepath: str) -> Graph:
    """
    安全地加载网络文件

    Args:
        filepath: 网络文件路径

    Returns:
        Graph 对象

    Raises:
        FileNotFoundError: 文件不存在
        ValueError: 文件格式错误
    """
    path = Path(filepath)

    # 检查文件是否存在
    if not path.exists():
        logger.error(f"Network file not found: {filepath}")
        raise FileNotFoundError(f"Network file not found: {filepath}")

    # 检查文件扩展名
    if path.suffix not in ['.net', '.paj']:
        logger.warning(f"Unexpected file extension: {path.suffix}")

    # 尝试加载文件
    try:
        logger.info(f"Loading network from {filepath}")
        graph = Graph.Read_Pajek(str(path))
        logger.info(f"Successfully loaded network with {graph.vcount()} vertices "
                   f"and {graph.ecount()} edges")
        return graph
    except Exception as e:
        logger.error(f"Failed to load network: {e}")
        raise ValueError(f"Failed to parse network file: {e}")
```

```python
# cli.py
import click
from network_analyzer import NetworkAnalyzer
import json

@click.group()
def cli():
    """网络分析工具 - 社区检测和可视化"""
    pass

@cli.command()
@click.argument('network_file', type=click.Path(exists=True))
@click.option('--method', '-m', default='walktrap',
              type=click.Choice(['walktrap', 'eigenvector', 'betweenness']),
              help='社区检测算法')
@click.option('--output', '-o', default='result.json',
              help='输出结果文件')
def detect(network_file, method, output):
    """检测网络中的社区结构"""
    click.echo(f"分析网络文件: {network_file}")
    click.echo(f"使用算法: {method}")

    analyzer = NetworkAnalyzer(network_file)
    result = analyzer.detect_communities(method)

    # 保存结果
    with open(output, 'w') as f:
        json.dump(result, f, indent=2)

    click.echo(f"✓ 检测到 {result['num_communities']} 个社区")
    click.echo(f"✓ 模块度: {result['modularity']:.4f}")
    click.echo(f"✓ 结果已保存到: {output}")

@cli.command()
@click.argument('network_file', type=click.Path(exists=True))
@click.option('--methods', '-m', multiple=True,
              default=['walktrap', 'eigenvector', 'betweenness'],
              help='要比较的算法')
def compare(network_file, methods):
    """比较多种社区检测算法"""
    click.echo(f"比较算法: {', '.join(methods)}")

    analyzer = NetworkAnalyzer(network_file)
    results = analyzer.compare_algorithms(list(methods))

    # 打印比较表格
    click.echo("\n算法比较结果:")
    click.echo("-" * 60)
    click.echo(f"{'算法':<15} {'社区数':<10} {'模块度':<15}")
    click.echo("-" * 60)

    for method, result in results.items():
        if result:
            click.echo(f"{method:<15} {result['num_communities']:<10} "
                      f"{result['modularity']:<15.4f}")
        else:
            click.echo(f"{method:<15} {'失败':<10}")

@cli.command()
@click.argument('network_file', type=click.Path(exists=True))
@click.option('--output', '-o', default='visualization.svg',
              help='输出图片文件')
@click.option('--layout', '-l', default='fr',
              type=click.Choice(['fr', 'circle', 'kk']),
              help='布局算法')
def visualize(network_file, output, layout):
    """可视化网络结构"""
    click.echo(f"可视化网络: {network_file}")

    analyzer = NetworkAnalyzer(network_file)
    analyzer.detect_communities('walktrop')  # 先检测社区
    analyzer.visualize(output_file=output, layout=layout)

    click.echo(f"✓ 可视化已保存到: {output}")

@cli.command()
@click.argument('network_file', type=click.Path(exists=True))
def stats(network_file):
    """显示网络统计信息"""
    analyzer = NetworkAnalyzer(network_file)
    statistics = analyzer.get_statistics()

    click.echo("\n网络统计信息:")
    click.echo("-" * 40)
    for key, value in statistics.items():
        if value is not None:
            click.echo(f"{key:<20}: {value}")
        else:
            click.echo(f"{key:<20}: N/A")

if __name__ == '__main__':
    cli()
```

```python
def process_large_network(network_file: str, chunk_size: int = 1000):
    """
    处理大规模网络（分块处理）

    Args:
        network_file: 网络文件路径
        chunk_size: 每次处理的节点数
    """
    import numpy as np

    # 使用内存映射读取大文件
    graph = Graph.Read_Pajek(network_file)

    # 如果网络太大，考虑采样
    if graph.vcount() > 10000:
        logger.warning(f"Large network detected ({graph.vcount()} vertices). "
                      "Consider sampling.")

        # 采样方法1：随机采样
        sample_vertices = np.random.choice(
            graph.vcount(),
            size=min(5000, graph.vcount()),
            replace=False
        )
        graph = graph.subgraph(sample_vertices)

    return graph
```
